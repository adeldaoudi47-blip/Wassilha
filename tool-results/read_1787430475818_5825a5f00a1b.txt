  1711→function MainNavigator({ route }: any) {
  1712→  const [role, setRole] = useState<Role>(route?.params?.role || 'customer');
  1713→  const { lang } = useContext(LanguageContext);
  1714→
  1715→  useEffect(() => {
  1716→    const check = async () => {
  1717→      const r = await AsyncStorage.getItem('wassilha_role');
  1718→      if (r) setRole(r as Role);
  1719→    };
  1720→    check();
  1721→    const interval = setInterval(check, 1000);
  1722→    return () => clearInterval(interval);
  1723→  }, []);
  1724→
  1725→  // Role switcher bar
  1726→  const RoleSwitcher = () => (
  1727→    <View style={[styles.roleSwitcherBar, { flexDirection: lang === 'ar' ? 'row-reverse' : 'row' }]}>
  1728→      <Text style={styles.roleSwitcherLabel}>{lang === 'ar' ? 'تجربة الأدوار:' : 'Rôles:'}</Text>
  1729→      {(['customer', 'driver', 'admin'] as Role[]).map((r) => (
  1730→        <TouchableOpacity
  1731→          key={r}
  1732→          onPress={async () => {
  1733→            setRole(r);
  1734→            await AsyncStorage.setItem('wassilha_role', r);
  1735→          }}
  1736→          style={[styles.roleSwitcherBtn, role === r && styles.roleSwitcherBtnActive]}
  1737→        >
  1738→          <Text style={[styles.roleSwitcherText, role === r && styles.roleSwitcherTextActive]}>
  1739→            {r === 'customer' ? (lang === 'ar' ? 'زبون' : 'Client') : r === 'driver' ? (lang === 'ar' ? 'سائق' : 'Chauffeur') : 'Admin'}
  1740→          </Text>
  1741→        </TouchableOpacity>
  1742→      ))}
  1743→    </View>
  1744→  );
  1745→
  1746→  return (
  1747→    <View style={{ flex: 1 }}>
  1748→      <RoleSwitcher />
  1749→      {role === 'customer' && <CustomerTabs />}
  1750→      {role === 'driver' && <DriverTabs />}
  1751→      {role === 'admin' && <AdminTabs />}
  1752→    </View>
  1753→  );
  1754→}
